Two versions are provided:

1. **Straight‑forward version** – builds a `set`, sorts it, and picks the second‑last element.  
   *Time*: `O(n log n)` because of the sort.  
   *Space*: `O(n)` for the intermediate set.

2. **Linear‑time version** – scans the list once, keeping track of the largest and second‑largest distinct values.  
   *Time*: `O(n)`; *Space*: `O(1)` (besides the input list).

Both functions share the same public signature and behaviour, so you can pick whichever you like.

```python
from typing import List, Optional


class NotEnoughUniqueNumbersError(ValueError):
    """Raised when < 2 distinct numbers are present in the input."""
    pass


def second_largest_unique_sort(nums: List[int]) -> int:
    """
    Return the second‑largest *unique* integer using sorting.

    Parameters
    ----------
    nums : List[int]
        Input list of integers (may contain duplicates).

    Returns
    -------
    int
        The second‑largest distinct value.

    Raises
    ------
    NotEnoughUniqueNumbersError
        If the list contains fewer than two distinct numbers.
    """
    # 1️⃣ Build the set of distinct numbers.
    unique = set(nums)
    # 2️⃣ We need at least two of them.
    if len(unique) < 2:
        raise NotEnoughUniqueNumbersError(
            "At least two distinct numbers are required."
        )
    # 3️⃣ Sort descending and pick the second element.
    #    `sorted` returns a list, so we can index it.
    sorted_desc = sorted(unique, reverse=True)
    return sorted_desc[1]


def second_largest_unique_linear(nums: List[int]) -> int:
    """
    Return the second‑largest *unique* integer in O(n) time and O(1) extra space.

    The algorithm keeps two variables, ``first`` (largest) and ``second`` (second‑largest),
    updating them while scanning the list once. Duplicate values are ignored.

    Parameters
    ----------
    nums : List[int]
        Input list of integers (may contain duplicates).

    Returns
    -------
    int
        The second‑largest distinct value.

    Raises
    ------
    NotEnoughUniqueNumbersError
        If the list contains fewer than two distinct numbers.
    """
    first: Optional[int] = None   # Largest seen so far
    second: Optional[int] = None  # Second‑largest seen so far

    for x in nums:
        # Skip if x has already been accounted for as first or second
        if x == first or x == second:
            continue

        if first is None or x > first:
            # New maximum found; demote the old maximum to second place
            second = first
            first = x
        elif second is None or x > second:
            # x is between first and second
            second = x
        # else: x is smaller than both first and second → ignore

    if second is None:
        # This means we never found a distinct value smaller than the maximum.
        raise NotEnoughUniqueNumbersError(
            "At least two distinct numbers are required."
        )
    return second

        ([-1, -3, -2, -4], -2),        # negative numbers

| Step | `x` | `first` before | `second` before | Action | `first` after | `second` after |
|------|-----|----------------|----------------|--------|---------------|----------------|
| 1    | 5   | `None`         | `None`         | `x > first` → update | 5 | `None` |
| 2    | 1   | 5              | `None`         | `x > second` → update | 5 | 1 |

def second_largest_unique(numbers):
    """
    Returns the second-largest unique number from a list of integers.
    
    Args:
        numbers (list): A list of integers
        
    Returns:
        int or None: The second-largest unique number, or None if there are 
                     fewer than two unique numbers
    """
    # Remove duplicates by converting to a set, then back to a list
    unique_numbers = list(set(numbers))
    
    # Handle case with fewer than 2 unique numbers
    if len(unique_numbers) < 2:
        return None
    
    # Sort in descending order and return the second element
    unique_numbers.sort(reverse=True)
    return unique_numbers[1]


# Alternative implementation without sorting (more efficient for large lists)
def second_largest_unique_v2(numbers):
    """
    Returns the second-largest unique number without sorting the entire list.
    More efficient for large lists (O(n) time complexity).
    """
    # Remove duplicates
    unique_numbers = set(numbers)
    
    # Handle case with fewer than 2 unique numbers
    if len(unique_numbers) < 2:
        return None
    
    # Find largest and second largest in one pass
    largest = second_largest = float('-inf')
    
    for num in unique_numbers:
        if num > largest:
            second_largest = largest
            largest = num
        elif num > second_largest:
            second_largest = num
    
    return second_largest

def second_largest_unique(numbers):
    """
    Return the second-largest unique integer from a list.
    
    If fewer than two unique numbers are provided, returns None.
    """
    unique_numbers = set(numbers)           # Remove duplicates
    if len(unique_numbers) < 2:             # Handle edge case
        return None
    
    # Sort descending and pick the second element
    sorted_unique = sorted(unique_numbers, reverse=True)
    return sorted_unique[1]


# --- Quick demos ---
print(second_largest_unique([3, 1, 4, 4, 2]))  # → 3
print(second_largest_unique([5, 5, 5]))       # → None
print(second_largest_unique([1, 2]))          # → 1
print(second_largest_unique([]))               # → None
```

**How it works:**
1. `set(numbers)` strips duplicates, leaving only unique values.
2. If there are fewer than 2 unique values, we return `None` immediately (the function "handles" the case gracefully instead of crashing).
3. Otherwise, we sort the unique values in descending order and return the element at index `1` (the second-largest).

**Complexity:** O(n log n) time due to sorting, O(n) space for the set. For very large lists, a single-pass O(n) scan is possible, but this version prioritizes readability and correctness.
